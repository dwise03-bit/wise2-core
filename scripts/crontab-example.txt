# WISE² Raspberry Pi Health Check - Crontab Examples
# Add these lines to your crontab using: crontab -e

# Hourly health check with email alert
0 * * * * /home/dwise/Projects/wise2-core/scripts/pi-health-check.sh --email >> /var/log/wise2-health-check.log 2>&1

# Alternative: Every 30 minutes (frequent monitoring)
# */30 * * * * /home/dwise/Projects/wise2-core/scripts/pi-health-check.sh >> /var/log/wise2-health-check.log 2>&1

# Alternative: Multiple checks with different schedules

# Quick check every 30 minutes (no email)
# */30 * * * * /home/dwise/Projects/wise2-core/scripts/pi-health-check.sh >> /var/log/wise2-health-check.log 2>&1

# Detailed check with email daily at 9 AM UTC
# 0 9 * * * /home/dwise/Projects/wise2-core/scripts/pi-health-check.sh --email --verbose >> /var/log/wise2-health-check.log 2>&1

# Weekly deep report (Monday 8 AM UTC)
# 0 8 * * 1 /home/dwise/Projects/wise2-core/scripts/pi-health-check.sh --email --verbose >> /var/log/wise2-health-check.log 2>&1

# With webhook to Discord (every 2 hours)
# 0 */2 * * * /home/dwise/Projects/wise2-core/scripts/pi-health-check.sh --webhook "https://discordapp.com/api/webhooks/YOUR_ID/YOUR_TOKEN" >> /var/log/wise2-health-check.log 2>&1

# With webhook and email (hourly)
# 0 * * * * /home/dwise/Projects/wise2-core/scripts/pi-health-check.sh --email --webhook "https://your-webhook-url" >> /var/log/wise2-health-check.log 2>&1

# Instructions:
# 1. Replace paths with actual paths to your installation
# 2. Update WEBHOOK URL if using webhook reporting
# 3. Ensure the scripts directory is readable by the cron user
# 4. Ensure /var/log directory is writable by the cron user
# 5. Test with: /home/dwise/Projects/wise2-core/scripts/pi-health-check.sh --verbose

# To install:
# 1. Copy desired lines above
# 2. Run: crontab -e
# 3. Paste lines
# 4. Save (Ctrl+X, Y in nano)
# 5. Verify: crontab -l
